<script>
  // Old acagamic.com (Hugo) addresses that no forwarding page covers go to the closest section.
  (() => {
    const path = location.pathname;
    const home = {
      "/newsletter": "/blog/",
      "/guides": "/blog/",
      "/tag": "/blog/",
      "/category": "/blog/",
      "/talks": "/teaching/",
      "/event": "/teaching/",
      "/publication": "/publications/",
    };
    const section = Object.keys(home).find((s) => path === s || path.startsWith(`${s}/`));
    if (section) location.replace(home[section]);
  })();
</script>
